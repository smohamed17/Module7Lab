-- this is focusing on the joining two tables method. 
SELECT c.CompanyName, o.OrderDate FROM Customers 
AS c JOIN Orders AS o ON c.CustomerID = o.CustomerID 
WHERE c.CompanyName = 'Around the Horn' LIMIT 5;
